{{- /* BlogPosting JSON-LD for pages carrying the `article` cascade (content/blog). Same idiom as schema-person.html: Go maps rendered with jsonify so quoting and escaping are Hugo's problem, not ours. `author` and `publisher` point at the Person node schema-person.html emits on the home page, via the same `#person` @id, so the two graphs join up. dateModified comes from .Lastmod, which resolves to .Date until a post sets `lastmod:` by hand. Deliberate: enableGitInfo would claim every post was revised on the date of the last theme refactor. */ -}} {{- $person := dict "@id" (printf "%s#person" site.BaseURL) -}} {{- $images := slice -}} {{- range .Params.images | default site.Params.images -}} {{- $images = $images | append (absURL .) -}} {{- end -}} {{- $article := dict "@context" "https://schema.org" "@type" "BlogPosting" "@id" (printf "%s#article" .Permalink) "headline" .Title "description" (.Description | default site.Params.description) "url" .Permalink "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "author" $person "publisher" $person "articleSection" (.Section | title) "wordCount" .WordCount "inLanguage" site.Language.Locale -}} {{- with $images -}} {{- $article = merge $article (dict "image" .) -}} {{- end -}} {{- with .Params.tags -}} {{- $article = merge $article (dict "keywords" .) -}} {{- end -}}
