{{- /* Countries visited, grouped by continent, from data/hobbies/countries.json. Countries flagged `home` are shown on the map but are not places visited. Each list is joined with `delimit` rather than printed inside the loop, so no template whitespace can leak in front of a comma or break a line. */}} {{- $continents := slice (dict "slug" "asia" "name" "Asia") (dict "slug" "europe" "name" "Europe") (dict "slug" "latin-america" "name" "Latin America") (dict "slug" "north-america" "name" "North America") (dict "slug" "africa" "name" "Africa") }}
{{- range $continents }} {{- $rows := where (where hugo.Data.hobbies.countries "continent" .slug) "home" "!=" true }} {{- $names := slice }} {{- range sort $rows "name" }}{{ $names = $names | append .name }}{{ end }}

{{ .name }}

{{ delimit $names ", " }}

{{- end }}
