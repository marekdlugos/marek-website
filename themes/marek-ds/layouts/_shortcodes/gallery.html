{{- /* Photo gallery shortcode: gallery width="wide|full" layout="justified|grid" caption="…". The inner content lists one image per line as "path | alt text"; the alt is optional. Images resolve against the page bundle first, then assets/images/. The justified layout sizes each item by its aspect ratio so rows fill the width with one to three photos; the grid layout is a plain two- or three-column grid. */}} {{- $width := .Get "width" | default "wide" }} {{- $layout := .Get "layout" | default "justified" }} {{- $items := slice }} {{- range split (trim .Inner "\n\r\t ") "\n" }} {{- $line := trim . " \t" }} {{- if $line }} {{- $parts := split $line "|" }} {{- $src := trim (index $parts 0) " " }} {{- $alt := "" }} {{- if gt (len $parts) 1 }}{{ $alt = trim (index $parts 1) " " }}{{ end }} {{- $img := or ($.Page.Resources.Get $src) (resources.Get (printf "images/%s" $src)) }} {{- if not $img }} {{- errorf "gallery: image %q not found in the page bundle or assets/images (%s)" $src $.Position }} {{- end }} {{- $items = $items | append (dict "img" $img "alt" $alt) }} {{- end }} {{- end }}
